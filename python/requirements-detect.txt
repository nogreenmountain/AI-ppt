# Runtime dependencies for the slide2pptx detector.
#
# Pillow + NumPy are the guaranteed baseline and are required at runtime.
# The remaining entries are optional -- the detector degrades gracefully
# when they are absent (warnings are recorded in ``detected.json``).

# --- Required baseline -----------------------------------------------------
Pillow==12.3.0
numpy==2.5.1

# --- Optional: OCR for native text elements --------------------------------
# When installed, ``slide2pptx.detect`` extracts OCR boxes / text /
# confidence and emits ``kind == "text"`` elements with the
# ``render_strategy == "native"``.  When missing, the detector still
# produces detected.json and the background PNG; only text elements are
# skipped.
rapidocr_onnxruntime==1.4.4 ; python_version >= "3.8" and python_version < "3.13"
rapidocr==3.9.2 ; python_version >= "3.13"
onnxruntime==1.28.0

# --- Optional: inpainting for the cleaned background -----------------------
# When installed, OCR regions are removed from the background and saved
# as ``cleaned-background.png``.  When missing, the original image is
# copied verbatim to ``original-background.png``.
opencv-python==5.0.0.93 ; python_version >= "3.8"
